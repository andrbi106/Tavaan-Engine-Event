## استفاده

```php
use Tavaan\Engine\Event\EventDispatcher;

$d = new EventDispatcher();

// Listen با Closure
$d->listen('user.created', function ($name) { /* ... */ });
$d->listen(OrderWasPaid::class, function (OrderWasPaid $e) { /* ... */ });

// Listen با priority
$d->listen('event', $highPriorityCallback, priority: 100);
$d->listen('event', $lowPriorityCallback, priority: 0);

// Wildcard
$d->listen('shop.order.*', function (...$args) { /* ... */ });

// Subscribe class
$d->subscribe(OrderEventSubscriber::class);

// Class-string listener (calls handle())
$d->listen('event', SendInvoiceListener::class);

// Class@method listener
$d->listen('event', SendInvoiceListener::class . '@handle');

// Array listener
$d->listen('event', [$instance, 'methodName']);

// Dispatch
$d->dispatch('user.created', ['Alice']);
$d->dispatch(new OrderWasPaid($order));

// Dispatch و توقف در اولین non-null (halt)
$result = $d->until('maintenance.check');

// Introspection
$d->hasListeners('user.created');          // bool
$d->getListeners('user.created');          // list<callable>
$d->forget('user.created');                // remove all

// Fake mode (برای تست)
$d->fake();
$d->dispatch('test.event');
$d->assertDispatched('test.event');
$d->assertDispatched('test.event', fn($payload) => /* ... */);
$d->assertNotDispatched('other.event');
$d->assertDispatchedTimes('test.event', 1);

// Fake با whitelist
$d->fake(['only.fake.this.event']);

// Clear fake (back to real)
$d->clearFake();
```
